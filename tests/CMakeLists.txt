# QTest ベースのテスト。1テストファイル = 1実行ファイルというシンプルな構成にする
# (QTEST_MAIN は1バイナリにつき1つの QObject テストクラスを前提とするため)。
function(boardes_add_test name)
	add_executable(${name} ${ARGN})
	target_link_libraries(${name} PRIVATE boardes_core Qt6::Widgets Qt6::Test)
	target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
	add_test(NAME ${name} COMMAND ${name})
	set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endfunction()
